cmake_minimum_required(VERSION 3.16)
project(g4proto_beamon_zero CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Geant4 REQUIRED)
include(${Geant4_USE_FILE})
add_executable(beamon_zero main.cc)
target_include_directories(beamon_zero PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../../../assets/g4sim)
target_link_libraries(beamon_zero ${Geant4_LIBRARIES})
